\begin{exercice}
    On considère l'application qui, à un vecteur du plan $\vec{v}$, associe le vecteur $-2\vec{v}$. Déterminer si cette application est linéaire.
\end{exercice}

\begin{exercice}
  Déterminer le produit des deux matrices suivantes :
  \[
    A
    =
    \begin{pmatrix}
      1 & 2 & -1 \\
      1 & 0 & 2 \\
      -1 & 2 & -1
    \end{pmatrix}
    \quad\text{et}\quad
    B
    =
    \begin{pmatrix}
      3 & 0 & 3 \\
      4 & 1 & 6 \\
      1 & 1 & 3
    \end{pmatrix}.
  \]
\end{exercice}

\begin{exercice}
  Montrer que tout vecteur $\vec{v}\in \mathbb{R}^2$ admet un unique antécédent par l'application linéaire
    \begin{equation*}
      f : \left \{
        \begin{array}{ccc}
          \mathbb{R}^2 &\longrightarrow &\mathbb{R}^2\\
          \dbinom{x}{y} &\longmapsto &
          \begin{pmatrix}
            2x + 3y \\
            x - 5y
          \end{pmatrix}.
        \end{array} \right.
      \end{equation*}
\end{exercice}
