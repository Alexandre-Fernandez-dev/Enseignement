    % Déterminer les entiers naturels $n$ tels que $5$ divise $4n - 3$.

    \begin{exercice}[10 min]
        \begin{enumerate}
            \item[]
            \item Déterminer l’ensemble des entiers naturels $n$ tels que $7n^2 + 2$ soit divisible par $3$.
            \item Déterminer l’ensemble des entiers relatifs $n$ tels que $3n + 4$ soit divisible par $4n + 3$.
        \end{enumerate}
    \end{exercice}
    
    \begin{exercice}[10 min]
        \begin{enumerate}
            \item[]
            \item Déterminer si $a$ est inversible modulo $n$ et déterminer l’inverse le cas échéant.
            \begin{tasks}(2)
                \task $a = 5$ et $n = 14$
                \task $a = 6$ et $n = 9$
            \end{tasks}
            \item Si possible, résoudre dans $\mathbb{Z}$ l'équation $5x -7 \equiv 0 \; [14]$.
        \end{enumerate}
    \end{exercice}
    
    \begin{exercice}[10 min]
        Énoncer la règle de divisibilité par $3$ puis la démontrer.
    \end{exercice}

    \begin{exercice}[15 min]
        Soient soit $n \in \mathbb{N}^\star$ et $S = \{a_0, \dots, a_n\}$ un ensemble de $n+1$ entiers naturels quelconques.
        Montrer qu'il doit exister deux entiers dans $S$ dont la différence est divisible par $n$.
    \end{exercice}
    
    \begin{exercice}[10 min]
        En informatique, un octet est composé de huit bits ($0$ ou $1$). Plus précisément, il est composé de sept bits contenant l’information à transmettre auquel on ajoute un bit de parité de la façon suivante : on calcule la somme $s$ des sept bits constituant l’octet et on prend $m$ tel que $s + m \equiv 0 \; [2]$.
        \begin{enumerate}
            \item Déterminer le bit de parité de $1011001$.
            %\item Démontrer que si un seul des sept bits contenant l’information est modifié, le bit de parité détecte l’erreur.
            \item Dans quel cas les erreurs de transmission seront-elles détectées ? Justifier.
        \end{enumerate}
    \end{exercice}
