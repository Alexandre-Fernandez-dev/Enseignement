\begin{exercice}
    \begin{enumerate}
        \item[]
        \item $T_1$ a pour équation $y = 2$:\\
        \\
        \underline{Justification :}\\
        $T_1$ est horizontale donc son coefficient directeur est $0$.\\
        On a donc $T_1: y = 0x + b$.\\
        Comme tout point de $T_1$ (ex: $A(0;\; 2)$) a pour ordonnée $2$ on a $b = 2$, en effet pour $A$:
        $$
            T_1(0) = 2 \iff b = 2
        $$
        Donc $T_1: y = 2$.
        \item $T_2$ a pour équation $y = -2x + 4$:\\
        \\
        \underline{Justification :}\\
        En prenant deux points par exemple $A\left(2;\; \dfrac{3}{2}\right)$ et $B\left(\dfrac{5}{2};\; \dfrac{1}{2}\right)$ on calcule le coefficient directeur:
        $$
            \frac{y_B - y_A}{x_B - x_A} = \dfrac{\dfrac{1}{2} - \dfrac{3}{2}}{\dfrac{5}{2} - 2} = \dfrac{1-3}{5 - 4} = -2
        $$
        Donc le coefficient directeur de $T_2$ est $f'(2) = -2$.\\
        L'équation de $T_2$ est donc $f'(2)(x-2) + f(2)$. Par lecture on trouve $f(2) = \dfrac{3}{2}$.
        Donc $T_2: y = -2(x-2) + \dfrac{3}{2} = -2x + \dfrac{11}{2}$.
    \end{enumerate}
\end{exercice}

\begin{exercice}
    \begin{enumerate}
        \item[]
        \item L'ensemble de dérivabilité de $f(x) = \dfrac{1}{x}$ est $\mathbb{R}_\star$. Comme $a = -5 \in \mathbb{R}_\star$, $f$ est bien dérivable en $a$.\\
        $f$ est la fonction inverse et a donc pour dérivée:
        $$
            f'(x) = -\frac{1}{x^2}
        $$
        Donc
        \begin{align*}
            f'(-5) &= -\frac{1}{(-5^2)}\\
            &= -\frac{1}{25}
        \end{align*}
        L'équation de la tangente est donc:
        \begin{align*}
            T_{-5}:\ & y = -\frac{1}{25}(x - (-5)) + f(-5)\\
            & y = -\frac{x}{25} - \frac{5}{25} - \frac{1}{5}\\
            & y = -\frac{x}{25} - \frac{2}{5}
        \end{align*}
        \item L'ensemble de dérivabilité de $f(x) = x^4$ est $\mathbb{R}$. Comme $a = 3 \in \mathbb{R}$, $f$ est bien dérivable en $a$.\\
        $f$ est la fonction puissance et a donc pour dérivée:
        $$
            f'(x) = 4x^3
        $$
        Donc
        \begin{align*}
            f'(3) &= 4\times3^3\\
            &= 4\times27\\
            &= 108
        \end{align*}
        L'équation de la tangente est donc:
        \begin{align*}
            T_3:\ & y = 108(x - 3) + f(3)\\
            & y = 108x - 324 + 81\\
        & y = 108x - 243
        \end{align*}
    \end{enumerate}
\end{exercice}


\begin{exercice}
    \begin{enumerate}
        \item[]
        \item Observons que la fonction $f(x) = \sqrt{1 - 2x}$ est composée: $f(x) = g(1- 2x)$ avec $g(x) = \sqrt{x}$.\\
        Comme la fonction racine est définie sur $\mathbb{R}_+$ et dérivable sur $\mathbb{R}_+^\star$. Il faut:
        \begin{itemize}
            \item Pour la définition: $1 - 2x \in \mathbb{R}_+$, soit $1 - 2x \geq 0$ et donc $x \in \left]-\infty;\; \dfrac{1}{2}\right]$.\\ Donc $f$ est définie sur $\left]-\infty;\; \dfrac{1}{2}\right]$.
            \item Pour la dérivabilité: if faut $1 - 2x \in \mathbb{R}_+^\star$, soit $1 - 2x > 0$ (cette fois on exclut $0$).\\
            De la même manière on trouve que $f$ est dérivable sur $\left]-\infty;\; \dfrac{1}{2}\right[$.
        \end{itemize}
        Calculons sa dérivée:
        \begin{align*}
            f'(x) &= a \times g'(ax + b) \qquad \text{(fonction composée)}\\
            &= -2 \times \dfrac{1}{2\sqrt{1-2x}}\\
            &= \dfrac{-1}{\sqrt{1 - 2x}}
        \end{align*}
        \item Considérons les différents termes de la fonction:
        \begin{itemize}
            \item $\dfrac{1}{x}$ n'est ni définie ni dérivable en $0$, ce qui est aussi le cas pour $\dfrac{7}{x}$.
            \item $x^3$ est définie et dérivable sur tout $\mathbb{R}$ donc c'est aussi le cas pour $\dfrac{x^3}{3}$.
        \end{itemize}
        Donc $f$ est définie et dérivable sur $\mathbb{R}^\star$.\\
        Calculons sa dérivée:
        \begin{align*}
            f'(x) &= \left(\frac{x^3}{3} + \frac{7}{x}\right)'\\
            &= \left(\frac{x^3}{3}\right)' + \left(\frac{7}{x}\right)' \qquad \text{(somme de fonction $(u+v)' = u' + v'$)}\\
            &= \frac{1}{3}\left(x^3\right)' + 7\left(\frac{1}{x}\right)' \qquad \text{(produit par une constante $(\lambda u)' = \lambda u')$}\\
            &= \frac{1}{3}\times 3x^2 + 7 \left(-\frac{1}{x^2}\right)\\
            &= x^2 - \frac{7}{x^2}
        \end{align*}
        \newpage
        \item Considérons les différents termes de la fonction:
        \begin{itemize}
            \item On a $\sqrt{x}$ dans la fonction qui est définie sur $\mathbb{R}_+$ et dérivable sur $\mathbb{R}_+^\star$.
            \item On a $x - 1$ au dénominateur. On doit donc avoir $x - 1 \neq 0$, soit $x \neq 1$.
        \end{itemize}
        Donc $f$ est définie sur $\mathbb{R}_+ \backslash \{ 1 \}$ et dérivable sur $\mathbb{R}_+^\star \backslash \{ 1 \}$ (ou $\mathbb{R}_+ \backslash \{ 0; 1 \}$).\\
        Pour calculer sa dérivée on observe que:
        $$
        f(x) = \frac{u}{v}
        $$
        avec
        \begin{align*}
        \begin{cases}
            u = \sqrt{x} & u' = \dfrac{1}{2\sqrt{x}}\\
            v = x - 1 & v' = 1
        \end{cases}
        \end{align*}
        Sa dérivée est donc:
        \begin{align*}
            f'(x) &= \frac{u'v - uv'}{v^2} \qquad \text{(produit de deux fonctions $(uv)'$)}\\
            &= \dfrac{\dfrac{1}{2\sqrt{x}}(x - 1) - \sqrt{x}\times 1}{(x - 1)^2}\\
            &= \frac{\dfrac{x}{2\sqrt{x}} - \dfrac{1}{2\sqrt{x}} - \dfrac{\sqrt{x}\times2\sqrt{x}}{2\sqrt{x}}}{(x - 1)^2}\\
            &= \frac{x-1-2\sqrt{x}^2}{2\sqrt{x}(x - 1)^2}\\
            &= \frac{x - 1 - 2x}{2\sqrt{x}(x - 1)^2}\\
            &= \frac{-x - 1}{2\sqrt{x}(x - 1)^2}
        \end{align*}
    \end{enumerate}
\end{exercice}

\begin{exercice}
    \begin{enumerate}
        \item[]
        \item Observons que $f(x)$ est le produit de deux fonctions:
        \begin{itemize}
            \item $(x+1)$ qui est définie et dérivable sur tout $\mathbb{R}$,
            \item $\sqrt(x+1)$ qui est définie pour $x + 1 \geq 0$ soit $x \geq -1$ et dérivable pour $x + 1 > 0$ soit $x > -1$.
        \end{itemize}
        Donc $f$ est définie sur $\left[-1; +\infty\right[$ (déjà donné dans l'énoncé) et dérivable sur $\left]-1; +\infty\right[$.
        \item Exprimons le taux d'accroissement en $-1$:
        \begin{align*}
            \frac{f(-1+h) - f(-1)}{h} &= \frac{\left[(-1+h)+1\right]\left(\sqrt{(-1 + h) + 1}\right) - \left[(-1 + 1)(\sqrt{-1 + 1})\right]}{h}\\
            &= \frac{h\sqrt{h}}{h}\\
            &= \sqrt{h}
        \end{align*}
        En prenant la limite pour $h \rightarrow 0$ on trouve $f'(-1) = 0$.\\
        Ainsi, la dérivée de $f$ est définie en $-1$, alors $f$ est dérivable en $-1$.\\
        Donc $f$ est dérivable sur $\left[-1; +\infty\right[$.
    \end{enumerate}
\end{exercice}
