\begin{exercice}
  \begin{enumerate}
    \item[]
    \item Soient trois points $A, B, C$ tels que \(\mathrm{AB} = 2, \mathrm{AC} = 2, \mathrm{BC} = 1\), déterminer $\overrightarrow{AB} \cdot \overrightarrow{BC}$.
      %\item Soient quatres points \(A(0\,;\,5), B(-7\,;\,8), C(1\,;\,5), D(-3\,;\,4)\). Calculer \(\overrightarrow{AB} \cdot \overrightarrow{CD}\).
    \item Soient \(A,\ B\) et \(C\) trois points tels que \(\mathrm{AB} = 1\), \(\mathrm{AC} = 5\) et \(\bigl(\overrightarrow{AB}, \overrightarrow{AC}\bigr) = -\dfrac{\pi}{4}\).
      Déterminer \(\overrightarrow{AB}\cdot\overrightarrow{AC}\).
  \end{enumerate}

\begin{exercice}
  Soient deux droites
  $$D_1 :\; 2x + 5y + 8 = 0 \quad \text{ et } \quad D_2 :\; 5x - 2y - 1 = 0$$
  Déterminer si $D_1$ et $D_2$ sont perpendiculaires.
\end{exercice}

\vfill

\begin{exercice}
  Dans chaque cas, indiquer si l'équation définit un cercle. Dans ce cas, déterminer les coordonnées du centre et le rayon du cercle défini par l'équation donnée.
  \begin{tasks}(2)
    \task $(\,x + 1\,)^2 + (\,y - 7\,)^2 = 1$,
    \task $5x^2 + 5y^2 - x + y = 87$.
  \end{tasks}
\end{exercice}

\vfill

\begin{exercice}
  Soient trois points $A, B$ et $C$ tels que
  $$
  \mathrm{AB} = 3, \quad \mathrm{AC} = 7 \quad \text{et} \quad \widehat{BAC} = 37^\circ
  $$
  Déterminer les trois longueurs des côtés ainsi que la mesure des trois angles du triangle $ABC$.
\end{exercice}
