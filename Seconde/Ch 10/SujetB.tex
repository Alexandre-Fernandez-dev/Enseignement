\begin{exercice}
  On donne ci-dessous le tableau de variation de deux fonctions $f$ et $g$.
  
  \noindent
  \begin{minipage}[t]{0.45\linewidth}
    \centering
    \begin{tikzpicture}
      \tkzTabInit[lgt=1.2,espcl=1.8]{$x$/0.8,$f(x)$/1.3}{-5,-3,0,3,5}
      \tkzTabVar{-/ -4,+/ 2,-/ -1,+/ 5,-/ 0}
    \end{tikzpicture}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.45\linewidth}
    \centering
    \begin{tikzpicture}
      \tkzTabInit[lgt=1.2,espcl=1.8]{$x$/0.8,$g(x)$/1.3}{-5,-2,1,4,5}
      \tkzTabVar{+/ 1,-/ -2,+/ 3,-/ 1,+/ 4}
    \end{tikzpicture}
  \end{minipage}

  \vspace{1ex}
  \begin{enumerate}[label=\arabic*.]
    \item Comparer, si c’est possible, les images suivantes avec un signe d’inégalité.
      \begin{tasks}(4)
        \task $f(-5)\;\rule{1em}{.15mm}\;f(-3)$
        \task $f(-1)\;\rule{1em}{.15mm}\;f(2)$
        \task $f(1)\;\rule{1em}{.15mm}\;f(3)$
        \task $f(0)\;\rule{1em}{.15mm}\;f(4)$
        \task $g(-4)\;\rule{1em}{.15mm}\;g(-2)$
        \task $g(2)\;\rule{1em}{.15mm}\;g(0)$
        \task $g(3)\;\rule{1em}{.15mm}\;g(5)$
        \task $g(0)\;\rule{1em}{.15mm}\;g(-4)$
      \end{tasks}

    \item Tracer une représentation possible des fonctions $f$ et $g$.
      \vspace{-1em}
      \begin{center}
        \begin{tikzpicture}[scale=3]
          \draw[step=.5cm,gray,thin] (-1.4,-1.4) grid (1.4,1.4);
          \draw[step=.5cm,gray,very thin, shift={(0.25, 0.25)}] (-1.65,-1.65) grid (1.15,1.15);
          \draw[step=.25cm,gray,very thin, dashed, shift={(0.125, 0.125)}] (-1.525,-1.525) grid (1.275,1.275);
          \draw[->] (-1.5,0) -- (1.5,0) node[right]{$x$};
          \draw[->] (0,-1.5) -- (0,1.5) node[above]{$y$};
          \foreach \x in {-5,...,5}{
            \draw (\x*0.25,1pt) -- (\x*0.25,-1pt) node[anchor=north,fill=white]{\small\x};
            \draw (1pt,\x*0.25) -- (-1pt,\x*0.25) node[anchor=east,fill=white]{\small\x};
          }
        \end{tikzpicture}
      \end{center}
  \end{enumerate}
\end{exercice}
