\section*{Exercice 1}
Déterminer si les fonctions suivantes sont croissantes ou décroissantes.
\begin{enumerate}
  \item $f(x)=5x-8$.\\
    Pour $x_1<x_2$, on a
    \[
      f(x_2)-f(x_1)=5(x_2-x_1)>0
    \]
    donc $f$ est \emph{strictement croissante} sur $\mathbb{R}$.
  \item $g(x)=-2x^2+6$.\\
    Pour $x_1<x_2$,
    \[
      g(x_2)-g(x_1)
      =-2(x_2^2-x_1^2)
      =-2(x_2-x_1)(x_2+x_1).
    \]
    Comme le signe de $x_2+x_1$ dépend des valeurs, on ne peut pas conclure un sens de variation unique sur $\mathbb{R}$.\\
    En fait,
    \[
      g'(x)=-4x,
    \]
    qui change de signe en $x=0$, donc $g$ n'est ni monotone croissante ni monotone décroissante sur tout $\mathbb{R}$.
\end{enumerate}

\section*{Exercice 3}
On donne le tableau de variation de deux fonctions $f$ et $g$.
\[
\begin{array}{c|ccccc}
x      & -5 & -2 & 1 & 4 & 5 \\ \hline
f(x)   & -3 & 0  & -2 & 5 & 1
\end{array}
\quad
\begin{array}{c|ccccc}
x      & -5 & 0 & 2 & 3 & 5 \\ \hline
g(x)   & 2  & -1 & 3 & 0 & 4
\end{array}
\]
\begin{enumerate}[label=\arabic*.]
  \item Comparer, si possible, les images suivantes :
  \begin{enumerate}
    \item $f(-4)\;<\;f(-3)$ car pour $x$ dans $[-5,-2]$, $f$ est croissante.
    \item $f(0)\;>\;f(1)$ car pour $x$ dans $[-2,1]$, $f$ est décroissante.
    \item $f(-2.5)\;<\;f(4.2)$ car $f(-2.5)\in[-5,0]$ croît jusqu'à $0$, puis décroît et recroît jusqu'à $5$ ; néanmoins $f(-2.5)<0<f(4.2)$.
    \item $f(-3.5)\;<\;f(2.1)$ car $f(-3.5)<0$ et $f(2.1)>0$.
    \item $g(1)\;<\;g(-4)$ car $g(-4)=2>g(0)=-1<g(1)=3$, donc $g(1)>g(-4)$.
    \item $g(0)\;<\;g(1)$ car sur $[-5,0]$, $g$ décroît de $2$ à $-1$, puis croît de $-1$ à $3$.
    \item $g(-4)\;>\;g(3.3)$ car $g(-4)=2>g(3.3)\approx0$.
    \item $g(1)\;>\;g(4)$ car $g(4)$ est entre $g(3)=0$ et $g(5)=4$, soit positif mais $g(1)=3<4$, donc $g(1)<g(4)$.
  \end{enumerate}
  \item Tracer une représentation possible de $f$ et $g$ sur les axes fournis correspondant aux tableaux ci-dessus.  
        (\emph{voir graphique apparié au tableau de variation})
\end{enumerate}

\section*{Exercice 4}
\begin{enumerate}
  \item $h(x)=x^2-8$.\\
    On reconnait une parabole d'équation $x^2+c$, de coefficient devant $x^2$ positif, donc $h$ admet un \emph{minimum} en son sommet $x=0$ et
    \[
      h(0)=0^2-8=-8.
    \]
  \item $k(x)=-4(x-2)^2+5$.\\
    C'est une parabole d'ouverture vers le bas (coefficient $-4<0$), donc $k$ admet un \emph{maximum} en $x=2$ de valeur
    \[
      k(2)=-4\cdot0^2+5=5.
    \]
\end{enumerate}
