\section*{Exercice 1}
On dispose des âges et effectifs suivants (total $N=24$) :
\[
\begin{array}{c|ccccc}
\text{Âge}       & 15 & 16 & 17 & 18 & 19 \\ \hline
\text{Effectif}  &  4 &  9 &  6 &  3 &  2
\end{array}
\]

\begin{enumerate}
  \item \textbf{Moyenne} 
    \[
      \bar{x} = \frac{1}{24}\sum n_i x_i
      = \frac{4\cdot15 + 9\cdot16 + 6\cdot17 + 3\cdot18 + 2\cdot19}{24}
      = \frac{398}{24}
      \approx 16{,}58.
    \]
    
  \item \textbf{Médiane}  
    Les effectifs cumulés sont :  
    \[
      15 :4,\quad 16:4+9=13,\quad 17:19,\quad 18:22,\quad 19:24.
    \]
    Les 12\up{e} et 13\up{e} valeurs se situent dans la classe d’âge 16, donc
    \[
      \mathrm{Med} = 16.
    \]

  \item \textbf{Quartiles}  
    \[
      Q_1 \text{ à la } 6,25\!e\text{ valeur}
      \Longrightarrow Q_1 = 16,
    \]
    \[
      Q_3 \text{ à la } 18,75\!e\text{ valeur}
      \Longrightarrow Q_3 = 17.
    \]

  \item \textbf{Écart interquartile}  
    \[
      IQR = Q_3 - Q_1 = 17 - 16 = 1.
    \]

  \item \textbf{Écart-type (population)}  
    Calcul des carrés :  
    \[
      \sum n_i x_i^2
      = 4\cdot15^2 + 9\cdot16^2 + 6\cdot17^2 + 3\cdot18^2 + 2\cdot19^2
      = 6632.
    \]
    \[
      \sigma
      = \sqrt{\frac{1}{N}\sum n_i x_i^2 \;-\;\bar{x}^{2}}
      = \sqrt{\frac{6632}{24} - \Bigl(\tfrac{398}{24}\Bigr)^{2}}
      \approx 1{,}14.
    \]
\end{enumerate}

\section*{Exercice 2}
On lit les effectifs depuis l'annexe~1 et on complète le tableau pour $N=23$ :

\[
\begin{array}{c|*{5}{c}|*{5}{c}}
\text{Note }x_i & 5 & 6 & 8 & 9 & 11 & 12 & 13 & 15 & 18 & 19 \\ \hline
n_i            & 1 & 3 & 6 & 2 &  1 &  4 &  2 &  2 &  1 &  1 \\ \hline
N_i            & 1 & 4 & 10 & 12 & 13 & 17 & 19 & 21 & 22 & 23 \\ \hline
f_i=\dfrac{n_i}{23}\,( \%) 
               & 4\% & 13\% & 26\% & 9\% & 4\% & 17\% & 9\% & 9\% & 4\% & 4\% \\ \hline
F_i \,( \%) 
               & 4\% & 17\% & 43\% & 52\% & 56\% & 73\% & 82\% & 91\% & 95\% & 99\% 
\end{array}
\]

\begin{enumerate}
  \item La \textbf{fréquence} de la note 6/20 est 
    \[
      f_6 = \frac{3}{23}\approx 0{,}1304 \approx 13\%.
    \]
    
  \item L’\textbf{effectif} des notes entre 8 et 13 inclus est
    \[
      6 + 2 + 1 + 4 + 2 = 15.
    \]

  \item La \textbf{fréquence} des notes $\ge15$ est
    \[
      \frac{2+1+1}{23} = \frac{4}{23}\approx 0{,}1739 \approx 17{,}4\%.
    \]

  \item \textbf{Médiane et quartiles (graphique)}  
    \[
      N=23,\quad \tfrac{N+1}{2}=12\Longrightarrow \mathrm{Med}=9,
    \]
    \[
      Q_1 \text{ à la } 6\!e\text{ valeur}\Longrightarrow Q_1=8,
    \quad
      Q_3 \text{ à la } 18\!e\text{ valeur}\Longrightarrow Q_3=13.
    \]
\end{enumerate}
