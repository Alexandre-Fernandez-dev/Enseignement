  \begin{enumerate}
    \item
      \begin{enumerate}
        \item Déterminer les coordonnées des vecteurs $\overrightarrow{AB}$ et $\overrightarrow{DC}$.
        \item Que peut-on en déduire ?
      \end{enumerate}
    \begin{center}
    \framebox[1\linewidth]{\parbox{0.92\linewidth}{
      \vspace{6em}
    }}
    \end{center}
    \item
      \begin{enumerate}
        \item Placer ces points dans le repère orthonormé ci-dessous.
        \item Déterminer les coordonnées de $\overrightarrow{AC}$:
        \begin{center}
          \framebox[1\linewidth]{\parbox{0.92\linewidth}{
            \vspace{2em}
          }}
          \end{center}
        \item Placer le point $E$ de sorte que $ADCE$ soit un parallélogramme.
        \item Déterminer les coordonnées de $E$ par le calcul.
        \begin{center}
        \framebox[1\linewidth]{\parbox{0.92\linewidth}{
          \vspace{9em}
        }}
        \end{center}
      \end{enumerate}
  \end{enumerate}
  \begin{center}
    \begin{tikzpicture}[scale=3.5]
      \draw[step=.5cm,gray,thin] (-1.4,-1.4) grid (1.4,1.4);

\begin{exercice}
  On considère les trois points suivants :
  $$A(3\,;2),\quad B(0\,;-2),\quad C(-6\,;2).$$

  On admet la propriété suivante :
  \emph{« Un quadrilatère $ABCD$ est un parallélogramme si et seulement si ses diagonales ont le même milieu ».}
  En utilisant cette propriété, déterminer les coordonnées du point $D$ qui fasse de $ABCD$ un parallélogramme.

\end{exercice}
