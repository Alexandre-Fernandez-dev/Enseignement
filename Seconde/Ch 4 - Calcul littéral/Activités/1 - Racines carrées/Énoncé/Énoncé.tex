\subsection*{Partie 1 : Rappel de la notion de racine carrée à travers les aires}

\begin{enumerate}
    \item \textbf{Construction de carrés :}
    \begin{enumerate}
        \item Dessinez des carrés d'aires $1~\text{cm}^2$, $4~\text{cm}^2$ et $9~\text{cm}^2$ sur du papier millimétré.
        \item Calculez et indiquez la longueur des côtés correspondants.
    \end{enumerate}
    \item \textbf{Carrés d'aires non entières :}
    \begin{enumerate}
        \item Dessinez un carré d'aire $2~\text{cm}^2$.
        \item Utilisez une règle et un compas pour construire ce carré.
        \item Indiquez la longueur du côté de ce carré en fonction de la racine carrée.
    \end{enumerate}
\end{enumerate}

\subsection*{Partie 2 : Lois de multiplication des racines carrées}

\begin{enumerate}
    \item \textbf{Multiplication géométrique :}
    \begin{enumerate}
        \item Construisez un rectangle dont les côtés mesurent $\sqrt{2}~\text{cm}$ et $\sqrt{8}~\text{cm}$.
        \item Calculez l'aire de ce rectangle.
    \end{enumerate}
    \item \textbf{Carré équivalent :}
    \begin{enumerate}
        \item Construisez un carré dont l'aire est égale à celle du rectangle précédent.
        \item Quelle est la longueur du côté de ce carré ?
        \item Comparez $\sqrt{2} \times \sqrt{8}$ et $\sqrt{16}$.
    \end{enumerate}
\end{enumerate}

\subsection*{Partie 3 : Exploration de l'inégalité $\sqrt{a + b} \leq \sqrt{a} + \sqrt{b}$}

\begin{enumerate}
    \item \textbf{Triangle rectangle :}
    \begin{enumerate}
        \item Prenez $a = 9$ et $b = 16$.
        \item Calculez $\sqrt{a}$, $\sqrt{b}$ et $\sqrt{a + b}$.
        \item Construisez un triangle rectangle dont les côtés perpendiculaires mesurent $\sqrt{a}~\text{cm}$ et $\sqrt{b}~\text{cm}$.
        \item Calculez la longueur de l'hypoténuse de ce triangle.
    \end{enumerate}
    \item \textbf{Analyse de l'inégalité :}
    \begin{enumerate}
        \item Calculez $\sqrt{a} + \sqrt{b}$.
        \item Comparez cette somme à $\sqrt{a + b}$.
        \item Que pouvez-vous conclure concernant l'inégalité $\sqrt{a + b} \leq \sqrt{a} + \sqrt{b}$ ?
    \end{enumerate}
\end{enumerate}
