    \begin{exercice}[title=Exercice 24 $\quad \star \quad$ { [Calculer] }]{}{}
    Dans chaque cas, déterminer si $a$ est inversible modulo $n$ et déterminer l’inverse le cas échéant.
    \begin{enumerate}
        \item $a = 3$ et $n = 7$
        \item $a = 3$ et $n = 10$
        \item $a = 4$ et $n = 12$
        \item $a = 6$ et $n = 9$
        \item $a = 9$ et $n = 16$
    \end{enumerate}
    \end{exercice}
    
    \begin{solutionbox}{Solution}
    Pour être inversible modulo $n$, il faut que $\gcd(a,n) = 1$.
    
    \begin{enumerate}
        \item $(a,n)=(3,7)$ :
    
        $\gcd(3,7)=1$, donc 3 est inversible modulo 7.
    
        Trouvons $x$ tel que $3x \equiv 1 \pmod{7}$. Par essais :
        $3 \times 5 =15 \equiv 1 \pmod{7}$.
    
        \textbf{Inverse de 3 mod 7: } 5.
    
        \item $(a,n)=(3,10)$ :
    
        $\gcd(3,10)=1$, donc 3 est inversible modulo 10.
    
        Trouvons $x$ tel que $3x \equiv 1 \pmod{10}$ :
        $3 \times 7=21 \equiv 1 \pmod{10}$.
    
        \textbf{Inverse de 3 mod 10 : } 7.
    
        \item $(a,n)=(4,12)$ :
    
        $\gcd(4,12)=4 \neq 1$, 4 n’est pas inversible modulo 12.
    
        \textbf{Pas d’inverse.}
    
        \item $(a,n)=(6,9)$ :
    
        $\gcd(6,9)=3 \neq 1$, 6 n’est pas inversible modulo 9.
    
        \textbf{Pas d’inverse.}
    
        \item $(a,n)=(9,16)$ :
    
        $\gcd(9,16)=1$, donc 9 est inversible modulo 16.
    
        Trouvons $x$ tel que $9x \equiv 1 \pmod{16}$ :
        $9 \times 9=81 \equiv 1 \pmod{16}$ (car 81-1=80 multiple de 16).
    
        \textbf{Inverse de 9 mod 16 : } 9.
    \end{enumerate}
    \end{solutionbox}

    \begin{exercice}[title=Exercice 25 $\quad \star \quad$ { [Calculer] }]{}{}
    Résoudre dans $\mathbb{Z}$ les équations suivantes :
    \begin{enumerate}
        \item $3x \equiv 1 \; [11]$
        \item $3x \equiv 4 \; [12]$
        \item $5x \equiv 7 \; [14]$
        \item $5x \equiv 6 \; [8]$
    \end{enumerate}
    \end{exercice}
    
    \begin{solutionbox}{Solution}
    \begin{enumerate}
        \item $3x \equiv 1 \pmod{11}$ :
    
        $\gcd(3,11)=1$, donc l’équation a une solution.
    
        Inverse de 3 modulo 11 est 4 (car $3 \times 4=12 \equiv 1 \pmod{11}$).
    
        $x \equiv 4 \times 1 = 4 \pmod{11}$.
    
        \textbf{Solution : } $x \equiv 4 \pmod{11}$.
    
        \item $3x \equiv 4 \pmod{12}$ :
    
        $\gcd(3,12)=3$. Pour qu’une solution existe, 3 doit diviser 4, ce n’est pas le cas.
    
        \textbf{Pas de solution.}
    
        \item $5x \equiv 7 \pmod{14}$ :
    
        $\gcd(5,14)=1$, donc l’inverse de 5 modulo 14 existe.
    
        Cherchons l’inverse de 5 mod14 :
        $5 \times 3=15 \equiv 1 \pmod{14}$.
    
        Inverse de 5 mod14: 3.
    
        $x \equiv 3 \times 7=21 \equiv 7 \pmod{14}$.
    
        \textbf{Solution : } $x \equiv 7 \pmod{14}$.
    
        \item $5x \equiv 6 \pmod{8}$ :
    
        $\gcd(5,8)=1$, inverse de 5 mod8 ?
        $5 \times 5=25 \equiv 1 \pmod{8}$, inverse:5.
    
        $x \equiv 5 \times 6=30 \equiv 6 \pmod{8}$.
    
        \textbf{Solution : } $x \equiv 6 \pmod{8}$.
    \end{enumerate}
    \end{solutionbox}
