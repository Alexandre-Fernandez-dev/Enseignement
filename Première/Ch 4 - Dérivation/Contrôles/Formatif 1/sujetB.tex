        \item $T_1$ a pour équation $y = 1$:\\
        \\
        \underline{Justification :}\\
        $T_1$ est horizontale donc son coefficient directeur est $0$.\\
        On a donc $T_1: y = 0x + b$.\\
        Comme tout point de $T_1$ (ex: $A(0; 1)$) a pour ordonnée $1$ on a $b = 1$, en effet pour $A$:
        $$
            T_1(0) = 1 \iff b = 1
        $$
        Donc $T_1: y = 1$.
        \item $T_2$ a pour équation $y = -2x + 4$:\\
        \\
        \underline{Justification :}\\
        En prenant deux points par exemple $A(\frac{3}{2}; 1)$ et $B(2; 0)$ on calcule le coefficient directeur:
        $$
            \frac{y_B - y_A}{x_B - x_A} = \frac{0 - -1}{2 - \frac{3}{2}} = \frac{-1}{\frac{1}{2}} = -2
        $$
        Donc le coefficient directeur de $T_2$ est $f'(2) = -2$.\\
        L'équation de $T_2$ est donc $f'(2)(x-2) + f(2)$. Par lecture on trouve $f(2) = 0$.
        Donc $T_2: y = -2(x-2) + 0 = -2x + 4$.
    \end{enumerate}
\end{exercice}

\begin{exercice}
    \begin{enumerate}
        \item[]
        \item L'ensemble de dérivabilité de $f(x) = \frac{1}{x}$ est $\mathbb{R}_\star$. Comme $a = -6 \in \mathbb{R}_\star$, $f$ est bien dérivable en $a$.\\
        $f$ est la fonction inverse et a donc pour dérivée:
        $$
            f'(x) = -\frac{1}{x^2}
        $$
        Donc
        \begin{align*}
            f'(-6) &= -\frac{1}{(-6^2)}\\
            &= -\frac{1}{36}
        \end{align*}
        L'équation de la tangente est donc:
        \begin{align*}
            T_{-6}:\ & y = -\frac{1}{36}(x - (-6)) + f(-6)\\
            & y = -\frac{x}{36} - \frac{6}{36} - \frac{1}{6}\\
            & y = -\frac{x}{36} - \frac{1}{3}
        \end{align*}
        \item L'ensemble de dérivabilité de $f(x) = x^3$ est $\mathbb{R}$. Comme $a = 5 \in \mathbb{R}$, $f$ est bien dérivable en $a$.\\
        $f$ est la fonction puissance et a donc pour dérivée:
        $$
            f'(x) = 3x^2
        $$
        Donc
        \begin{align*}
            f'(5) &= 3\times5^2\\
            &= 3\times25\\
            &= 75
        \end{align*}
        L'équation de la tangente est donc:
        \begin{align*}
            T_5:\ & y = 75(x - 5) + f(5)\\
            & y = 75x - 375 + 125\\
            & y = 75x - 250
        \end{align*}
    \end{enumerate}
\end{exercice}

\begin{exercice}
    \begin{enumerate}
        \item[]
        \item Considérons les différents termes de la fonction:
        \begin{itemize}
            \item $x^4$ est définie et dérivable sur tout $\mathbb{R}$ donc c'est aussi le cas pour $\frac{x^4}{4}$.
            \item $\frac{1}{x}$ n'est ni définie ni dérivable en $0$, ce qui est aussi le cas pour $\frac{5}{x}$.
        \end{itemize}
        Donc $f$ est définie et dérivable sur $\mathbb{R}^\star$.\\
        Calculons sa dérivée:
        \begin{align*}
            f'(x) &= \left(-\frac{x^4}{4} + \frac{5}{x}\right)'\\
            &= \left(-\frac{x^4}{4}\right)' + \left(\frac{5}{x}\right)' \qquad \text{(somme de fonction $(u+v)' = u' + v'$)}\\
            &= -\frac{1}{4}\left(x^4\right)' + 5\left(\frac{1}{x}\right)' \qquad \text{(produit par une constante $(\lambda u)' = \lambda u')$}\\
            &= -\frac{1}{4}\times 4x^3 + 5 \left(-\frac{1}{x^2}\right)\\
            &= -x^3 - \frac{5}{x^2}
        \end{align*}
        \item Considérons les différents termes de la fonction:
        \begin{itemize}
            \item On a $\sqrt{x}$ dans la fonction qui est définie sur $\mathbb{R}_+$ et dérivable sur $\mathbb{R}_+^\star$.
            \item On a $x + 1$ au dénominateur. On doit donc avoir $x + 1 \neq 0$, soit $x \neq - 1$.
        \end{itemize}
        Donc $f$ est définie sur $\mathbb{R}_+$ et dérivable sur $\mathbb{R}_+^\star$ (ou $\mathbb{R}_+ \backslash \{ 0 \}$).\\
        Pour calculer sa dérivée on observe que:
        $$
        f(x) = \frac{u}{v}
        $$
        avec
        \begin{align*}
        \begin{cases}
            u = \sqrt{x} & u' = \frac{1}{2\sqrt{x}}\\
            v = x + 1 & v' = 1
        \end{cases}
        \end{align*}
        Sa dérivée est donc:
        \begin{align*}
            f'(x) &= \frac{u'v - uv'}{v^2} \qquad \text{(produit de deux fonctions $(uv)'$)}\\
            &= \frac{\frac{1}{2\sqrt{x}}(x + 1) - \sqrt{x}\times1}{(x + 1)^2}\\
            &= \frac{\frac{\sqrt{x}}{2x}(x+1) - 2x\frac{\sqrt{x}}{2x}}{(x + 1)^2}\\
            &= \frac{\frac{\sqrt{x}}{2x}(x + 1 - 2x)}{(x+1)^2}\\
            &= \frac{\sqrt{x}(1 - x)}{2x(x+1)^2}\\
            &= \frac{1 - x}{2\sqrt{x}(x + 1)^2}
        \end{align*}
        \item Observons que la fonction $f(x) = \sqrt{-2x + 1}$ est composée: $f(x) = g(-2x + 1)$ avec $g(x) = \sqrt{x}$.\\
        Comme la fonction racine est définie sur $\mathbb{R}_+$ et dérivable sur $\mathbb{R}_+^\star$. Il faut:
        \begin{itemize}
            \item Pour la définition: $-2x + 1 \in \mathbb{R}_+$, soit $-2x + 1 \geq 0$ et donc $x \in \left]-\infty; \frac{1}{2}\right]$.\\ Donc $f$ est définie sur $\left]-\infty; \frac{1}{2}\right]$.
            \item Pour la dérivabilité: if faut $-2x + 1 \in \mathbb{R}_+^\star$, soit $-2x + 1 > 0$.\\
            De la même manière on trouve que $f$ est dérivable sur $\left]-\infty; \frac{1}{2}\right[$ (cette fois on exclut $0$).
        \end{itemize}
        Calculons sa dérivée:
        \begin{align*}
            f'(x) &= a \times g'(ax + b) \qquad \text{(fonction composée)}\\
            &= -2 \times \dfrac{1}{2\sqrt{-2x+1}}\\
            &= \dfrac{-1}{\sqrt{-2x + 1}}
        \end{align*}
    \end{enumerate}
\end{exercice}

\begin{exercice}
    \begin{enumerate}
        \item[]
        \item Observons que $f(x)$ est le produit de deux fonctions:
        \begin{itemize}
            \item $(x+1)$ qui est définie et dérivable sur tout $\mathbb{R}$,
            \item $\sqrt(x+1)$ qui est définie pour $x + 1 \geq 0$ soit $x \geq -1$ et dérivable pour $x + 1 > 0$ soit $x > -1$.
        \end{itemize}
        Donc $f$ est définie sur $\left[-1; +\infty\right[$ (déjà donné dans l'énoncé) et dérivable sur $\left]-1; +\infty\right[$.
        \item Exprimons le taux d'accroissement en $-1$:
        \begin{align*}
            \frac{f(-1+h) - f(-1)}{h} &= \frac{\left[(-1+h)+1\right]\left(\sqrt{(-1 + h) + 1}\right) - \left[(-1 + 1)(\sqrt{-1 + 1})\right]}{h}\\
            &= \frac{h\sqrt{h}}{h}\\
            &= \sqrt{h}
        \end{align*}
        En prenant la limite pour $h \rightarrow 0$ on trouve $f'(-1) = 0$.\\
        Ainsi, la dérivée de $f$ est définie en $-1$, alors $f$ est dérivable en $-1$.\\
        Donc $f$ est dérivable sur $\left[-1; +\infty\right[$.
    \end{enumerate}
\end{exercice}
