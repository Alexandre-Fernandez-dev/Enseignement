\section*{Exercice 1}
\begin{enumerate}
  \item Soient trois points $A,B,C$ tels que $AB=3$, $AC=4$, $BC=2$. Déterminer $\overrightarrow{AB}\cdot\overrightarrow{BC}$.

    On utilise l'expansion :
  \[
  \|\overrightarrow{AC}\|^2
  = \|\overrightarrow{AB} + \overrightarrow{BC}\|^2
  = \|\overrightarrow{AB}\|^2 + \|\overrightarrow{BC}\|^2 + 2\,\overrightarrow{AB}\cdot\overrightarrow{BC}.
  \]
  D'où
  \[
  2\,\overrightarrow{AB}\cdot\overrightarrow{BC}
  = AC^2 - AB^2 - BC^2
  = 4^2 - 3^2 - 2^2
  = 16 - 9 - 4
  = 3,
  \]
  et donc
  \[
  \boxed{\overrightarrow{AB}\cdot\overrightarrow{BC} = \frac{3}{2}.}
\]


  \item Soient $A,B,C$ tels que $AB=2$, $AC=6$ et $\widehat{(AB,AC)}=\frac\pi3$. Calculer $\overrightarrow{AB}\cdot\overrightarrow{AC}$.

    Rappel :
    \[
      \overrightarrow{AB}\cdot\overrightarrow{AC}
      = AB\;\times AC\;\times\cos\Bigl(\widehat{(AB,AC)}\Bigr).
    \]
    Donc
    \[
      \overrightarrow{AB}\cdot\overrightarrow{AC}
      = 2\cdot6\cdot\cos\frac\pi3
      = 12\cdot\frac12
      = \boxed{6.}
    \]
\end{enumerate}

\section*{Exercice 2}
On considère les droites
\[
  D_1 : 3x + 4y - 5 = 0
  \quad\text{et}\quad
  D_2 : 4x - 3y + 2 = 0.
\]
Le vecteur normal à $D_1$ est $n_1=(3,4)$ et celui à $D_2$ est $n_2=(4,-3)$. Elles sont perpendiculaires si et seulement si
\[
  n_1\cdot n_2 = 3\cdot4 + 4\cdot(-3) = 12 -12 = 0.
\]
Donc
\[
  \boxed{D_1 \perp D_2.}
\]

\section*{Exercice 3}
Indiquer si chaque équation définit un cercle. Le cas échéant, donner centre et rayon.
\begin{enumerate}
  \item $(x-2)^2 + (y+3)^2 = 9$.

    C’est déjà sous forme canonique $(x-\alpha)^2+(y-\beta)^2=R^2$.
    \[
      \boxed{\text{Centre }(2,-3),\quad \text{rayon }3.}
    \]

  \item $4x^2 + 4y^2 -2x +8y = 44$.

    On divise par 4 :
    \[
      x^2 + y^2 - \tfrac12 x +2y = 11.
    \]
    Complétons le carré :
    \[
      x^2 - \tfrac12 x = \bigl(x - \tfrac14\bigr)^2 - \tfrac1{16},
      \quad
      y^2 + 2y = (y+1)^2 -1.
    \]
    Donc
    \[
      \bigl(x - \tfrac14\bigr)^2 -\tfrac1{16}
      + (y+1)^2 -1 = 11
      \quad\Longrightarrow\quad
      \bigl(x - \tfrac14\bigr)^2 + (y+1)^2 = 12 + \tfrac1{16}
      = \tfrac{193}{16}.
    \]
    \[
      \boxed{\text{Centre }\bigl(\tfrac14,-1\bigr),\quad
      \text{rayon }\tfrac{\sqrt{193}}4.}
    \]
\end{enumerate}

\section*{Exercice 4}
Soient $A,B,C$ tels que
\[
  AB=4,\quad AC=5,\quad \widehat{BAC}=45^\circ.
\]
\begin{enumerate}
  \item {\bf Longueur $BC$ (loi des cosinus)} :
    \[
      BC^2 = AB^2 + AC^2 -2\,AB\cdot AC\cos\,45^\circ
      = 4^2 +5^2 -2\cdot4\cdot5\cdot\frac{\sqrt2}{2}
      = 41 -20\sqrt2.
    \]
    \[
      \boxed{BC = \sqrt{41 -20\sqrt2}.}
    \]

  \item {\bf Angles $B$ et $C$ (loi des sinus)} :

    Loi des sinus :
    \[
      \frac{\sin A}{BC} = \frac{\sin B}{AC} = \frac{\sin C}{AB}.
    \]
    On a $A=45^\circ$, $AC=5$, $AB=4$, $BC=\sqrt{41-20\sqrt2}$.  
    \[
      \sin B = \frac{AC}{BC}\,\sin A
      = \frac{5}{\sqrt{41-20\sqrt2}}\cdot\frac{\sqrt2}{2},
      \quad B\approx79{,}0^\circ.
    \]
    \[
      C = 180^\circ - A - B \approx 180 -45 -79 = 56^\circ.
    \]
    \[
      \boxed{A=45^\circ,\;B\approx58^\circ,\;C\approx77^\circ.}
    \]
\end{enumerate}

\section*{Exercice 5}
\begin{enumerate}
  \item Simplifier :
    \[
      \frac{e^{2x}\,(e^{4x})^2}{e^{10x}}
      = \frac{e^{2x}\,e^{8x}}{e^{10x}}
      = 1.
    \]
  \item Résoudre :
    \begin{enumerate}
      \item $e^{x+2} = e^{2x-1}$.
        \[
          x+2 = 2x-1
          \;\Longrightarrow\;
          x = 3.
        \]
        \[
          \boxed{x=3.}
        \]

      \item $e^x + 2 = 0$.

        Pour tout réel $x$, $e^x>0$, donc $e^x+2>2>0$.  
        \[
          \boxed{\text{Pas de solution réelle.}}
        \]

      \item $e^{-2x+3} < \dfrac1{e^{-x^2}}$.

        Or $\dfrac1{e^{-x^2}} = e^{x^2}$, donc
        \[
          e^{-2x+3} < e^{x^2}
          \;\Longleftrightarrow\;
          -2x+3 < x^2
          \;\Longleftrightarrow\;
          x^2 +2x -3 >0
          \;\Longleftrightarrow\;
          (x+3)(x-1)>0.
        \]
        \[
          \boxed{(-\infty,-3)\,\cup\,(1,+\infty).}
        \]
    \end{enumerate}
\end{enumerate}

\section*{Exercice 6}
\begin{enumerate}
  \item \textbf{Dériver} :
    \begin{enumerate}
      \item $f(x)=e^{2x}-3x^2+1$.
        \[
          f'(x) = 2e^{2x} -6x.
        \]

      \item $g(x)=e^{-x+4}$.
        \[
          g'(x) = -\,e^{-x+4}.
        \]
    \end{enumerate}

  \item \textbf{Étude de $h(x)=(2+x)e^{-x}$}.  

    \[
      h'(x) = \bigl(2+x\bigr)'\,e^{-x} + (2+x)\,(e^{-x})'
             = e^{-x} - (2+x)e^{-x}
             = -(x+1)e^{-x}.
    \]
    Comme $e^{-x}>0$, le signe de $h'(x)$ est celui de $-(x+1)$ :
    \[
      \begin{cases}
        h'(x)>0 &\iff x+1<0 \iff x<-1,\\
        h'(x)=0 &\iff x=-1,\\
        h'(x)<0 &\iff x>-1.
      \end{cases}
    \]
    \bigskip

    \noindent\textbf{Tableau de variations} :
    \[
      \begin{array}{c|ccc|c}
         x        & -\infty   &    -1      & +\infty \\ \hline
         h'(x)    &    +      &     0      &   -     \\ \hline
         h(x)     & \nearrow  & h(-1)=e   & \searrow
      \end{array}
    \]
    Ainsi, $h$ est croissante sur $(-\infty,-1]$, atteint son maximum $e$ en $x=-1$, puis décroît sur $[-1,+\infty)$.
\end{enumerate}
