\section*{Exercice 1}
\begin{enumerate}
  \item Soient \(A,B,C\) tels que \(\mathrm{AB}=5\), \(\mathrm{AC}=3\), \(\mathrm{BC}=4\). Calculer 
  \(\overrightarrow{AB}\cdot\overrightarrow{BC}\).
  
  On utilise l'identité dans le triangle \(ABC\) :
  \[
    \|\overrightarrow{AB} + \overrightarrow{BC}\|^2
    = AC^2
    = \|\overrightarrow{AB}\|^2 + \|\overrightarrow{BC}\|^2 + 2\,\overrightarrow{AB}\cdot\overrightarrow{BC}.
  \]
  Ainsi
  \[
    2\,\overrightarrow{AB}\cdot\overrightarrow{BC}
    = AC^2 - AB^2 - BC^2
    = 3^2 - 5^2 - 4^2
    = 9 - 25 - 16
    = -32,
  \]
  d'où
  \[
    \boxed{\overrightarrow{AB}\cdot\overrightarrow{BC} = -16.}
  \]
  
  \item Soient \(A,B,C\) tels que \(\mathrm{AB}=3\), \(\mathrm{AC}=7\) et 
  \(\bigl(\overrightarrow{AB},\overrightarrow{AC}\bigr)=\tfrac{2\pi}{3}\). 
  Trouver \(\overrightarrow{AB}\cdot\overrightarrow{AC}\).
  
  Par définition du produit scalaire :
  \[
    \overrightarrow{AB}\cdot\overrightarrow{AC}
    = AB\;\times AC\;\times \cos\!\Bigl(\tfrac{2\pi}{3}\Bigr)
    = 3\cdot7\cdot\bigl(-\tfrac12\bigr)
    = \boxed{-\tfrac{21}{2}}.
  \]
\end{enumerate}

\section*{Exercice 2}
On considère les droites  
\[
D_1:5x-2y+3=0
\quad\text{et}\quad
D_2:2x+5y-4=0.
\]
Le vecteur normal à \(D_1\) est \(n_1=(5,-2)\) et celui à \(D_2\) est \(n_2=(2,5)\).  
On calcule :
\[
n_1 \cdot n_2 = 5\cdot2 + (-2)\cdot5 = 10 -10 = 0.
\]
Donc 
\[
\boxed{D_1 \perp D_2.}
\]

\section*{Exercice 3}
Vérifier pour chaque équation si c’est un cercle et, le cas échéant, donner centre et rayon.
\begin{enumerate}
  \item \((x+4)^2+(y-1)^2=16\).\\
    C’est déjà la forme canonique \((x-\alpha)^2+(y-\beta)^2=R^2\) avec  
    \[
      \alpha=-4,\quad \beta=1,\quad R=\sqrt{16}=4.
    \]
    \[
      \boxed{\text{Centre }(-4,1),\quad\text{rayon }4.}
    \]
  
  \item \(9x^2+9y^2+6x-12y=27\).\\
    On divise par 9 :
    \[
      x^2+y^2+\tfrac{2}{3}x - \tfrac{4}{3}y = 3.
    \]
    On complète le carré :
    \[
      x^2 + \tfrac{2}{3}x
      = \Bigl(x + \tfrac{1}{3}\Bigr)^2 - \tfrac{1}{9},
      \quad
      y^2 - \tfrac{4}{3}y
      = \Bigl(y - \tfrac{2}{3}\Bigr)^2 - \tfrac{4}{9}.
    \]
    D'où
    \[
      \Bigl(x + \tfrac{1}{3}\Bigr)^2 - \tfrac{1}{9}
      + \Bigl(y - \tfrac{2}{3}\Bigr)^2 - \tfrac{4}{9}
      = 3
      \quad\Longrightarrow\quad
      \Bigl(x + \tfrac{1}{3}\Bigr)^2 + \Bigl(y - \tfrac{2}{3}\Bigr)^2
      = 3 + \tfrac{5}{9}
      = \tfrac{32}{9}.
    \]
    \[
      \boxed{\text{Centre }\bigl(-\tfrac{1}{3},\tfrac{2}{3}\bigr),\quad
      \text{rayon }\sqrt{\tfrac{32}{9}} = \tfrac{4\sqrt2}{3}.}
    \]
\end{enumerate}

\section*{Exercice 4}
Dans le triangle \(ABC\) on a
\[
AB=6,\quad AC=4,\quad \widehat{BAC}=30^\circ.
\]
\begin{enumerate}
  \item \textbf{Longueur \(BC\)} (loi des cosinus) :
    \[
      BC^2 = AB^2 + AC^2 - 2\,AB\cdot AC \cos 30^\circ
      = 6^2 + 4^2 - 2\cdot6\cdot4\cdot\frac{\sqrt3}{2}
      = 52 - 24\sqrt3.
    \]
    \[
      \boxed{BC = \sqrt{52 - 24\sqrt3}.}
    \]
  \item \textbf{Angles \(B\) et \(C\)} (loi des sinus) :\\
    \[
      \frac{\sin 30^\circ}{BC} = \frac{\sin B}{AC}
      \quad\Longrightarrow\quad
      \sin B = \frac{AC}{BC}\,\sin30^\circ
             = \frac{4}{\sqrt{52-24\sqrt3}}\cdot\tfrac12
             = \frac{2}{\sqrt{52-24\sqrt3}}.
    \]
    On en déduit \(B\approx38{,}2^\circ\).  
    Puis 
    \[
      C = 180^\circ - 30^\circ - B \approx 111{,}8^\circ.
    \]
\end{enumerate}

\section*{Exercice 5}
\begin{enumerate}
  \item Simplifier 
  \(\displaystyle \frac{e^{3x}\,(e^{2x})^3}{e^{11x}}\).\\
  On a \((e^{2x})^3 = e^{6x}\), donc
  \[
    \frac{e^{3x}\,e^{6x}}{e^{11x}}
    = e^{3x+6x-11x}
    = e^{-2x}.
  \]
  \[
    \boxed{e^{-2x}.}
  \]
  \item Résoudre :
  \begin{tasks}(3)
    \task \(e^{2x}-e^x=0\).\\
      \(e^x(e^x-1)=0\) et \(e^x\neq0\), donc \(e^x=1\), soit \(x=0\).
    \task \(e^x-1=0\).\\
      \(e^x=1\), donc \(x=0\).
    \task \(e^{-x+2}>\dfrac1{e^{-x^2}}\).\\
      \(\dfrac1{e^{-x^2}}=e^{x^2}\), l’inégalité devient
      \[
        e^{-x+2}>e^{x^2}
        \;\Longleftrightarrow\;
        -x+2 > x^2
        \;\Longleftrightarrow\;
        x^2 + x - 2 < 0
        \;\Longleftrightarrow\;
        (x+2)(x-1)<0.
      \]
      \[
        \boxed{-2 < x < 1.}
      \]
  \end{tasks}
\end{enumerate}

\section*{Exercice 6}
\begin{enumerate}
  \item Dériver :
  \begin{tasks}(2)
    \task \(f(x)=e^{x}+x^3-2\).\\
      \[
        f'(x)=e^x + 3x^2.
      \]
    \task \(g(x)=e^{3x-1}\).\\
      \[
        g'(x)=3\,e^{3x-1}.
      \]
  \end{tasks}
  \item Étudier le signe et les variations de 
  \(\displaystyle h(x)=(3+x)\,e^{-2x+1}\).\\
  On calcule
  \[
    h'(x)
    = (3+x)'\,e^{-2x+1} + (3+x)\,(e^{-2x+1})'
    = e^{-2x+1} -2(3+x)\,e^{-2x+1}
    = -(2x+5)\,e^{-2x+1}.
  \]
  Comme \(e^{-2x+1}>0\), le signe de \(h'(x)\) est celui de \(-(2x+5)\) :
  \[
    \begin{cases}
      h'(x)>0 &\iff 2x+5<0 \iff x<-\tfrac{5}{2},\\
      h'(x)=0 &\iff x=-\tfrac{5}{2},\\
      h'(x)<0 &\iff x>-\tfrac{5}{2}.
    \end{cases}
  \]
  \medskip

  \noindent\textbf{Tableau de variations :}
  \[
    \begin{array}{c|ccc|c}
      x        & -\infty      & -\tfrac{5}{2} & +\infty \\ \hline
      h'(x)    & +            & 0             & -       \\ \hline
      h(x)     & \nearrow     & \displaystyle \max\Bigl(h(-\tfrac{5}{2})\Bigr)
                  & \searrow
    \end{array}
  \]
  On calcule \(h(-\tfrac{5}{2}) = \bigl(3 - \tfrac{5}{2}\bigr)\,e^{-2(-5/2)+1}
    = \tfrac12\,e^6.\)
\end{enumerate}
