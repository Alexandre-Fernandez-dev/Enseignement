\subsection*{Correction : calculer une probabilité conditionnelle à l'aide d'un tableau}

\begin{enumerate}
    \item
    \begin{enumerate}
        \item
        La probabilité qu'un patient soit traité avec le médicament A est égale à :
        
        $$
        P(A) = \frac{455}{800} \approx 0{,}57 = 57\,\%.
        $$

        \item
        La probabilité qu'un patient soit guéri est égale à :
        
        $$
        P(G) = \frac{674}{800} \approx 0{,}84 = 84\,\%.
        $$

        \item
        La probabilité qu'un patient soit guéri et qu'il soit traité par le médicament A est égale à :
        
        $$
        P(G \cap A) = \frac{383}{800} \approx 0{,}48 = 48\,\%.
        $$

        \item
        La probabilité qu'un patient ne soit pas guéri et qu'il soit traité par le médicament A est égale à :
        
        $$
        P(\overline{G} \cap A) = \frac{72}{800} \approx 0{,}09 = 9\,\%.
        $$
    \end{enumerate}

    \item
    \begin{enumerate}
        \item
        \begin{center}
        \begin{tabular}{|c|c|c|c|}
        \hline
        & Médicament A & Médicament B & Total \\
        \hline
        \rowcolor{cyan!70}
        \textcolor{white}{Guéri} & \textcolor{white}{383} & \textcolor{white}{291} & \textcolor{white}{674} \\
        \hline
        Non guéri & 72 & 54 & 126 \\
        \hline
        Total & 455 & 345 & 800 \\
        \hline
        \end{tabular}
        \end{center}

        La probabilité que le patient ait pris le médicament A \textbf{sachant qu'il est guéri}
        se note $P_G(A)$ et est égale à :

        $$
        P_G(A) = \frac{383}{674} \approx 0{,}57 = 57\,\%.
        $$

        On regarde uniquement \textcolor{cyan}{la ligne des patients guéris}.

        \item
        \begin{center}
        \begin{tabular}{|c|c|c|c|}
        \hline
        & Médicament A & \cellcolor{orange!80}Médicament B & Total \\
        \hline
        Guéri & 383 & \cellcolor{orange!80}291 & 674 \\
        \hline
        Non guéri & 72 & \cellcolor{orange!80}54 & 126 \\
        \hline
        Total & 455 & \cellcolor{orange!80}345 & 800 \\
        \hline
        \end{tabular}
        \end{center}

        La probabilité que le patient soit guéri \textbf{sachant qu'il a pris le médicament B}
        se note $P_A(G)$ et est égale à :

        $$
        P_A(G) = \frac{291}{345} \approx 0{,}84 = 84\,\%.
        $$

        On regarde uniquement \textcolor{orange}{la colonne du médicament B}.
    \end{enumerate}
\end{enumerate}
