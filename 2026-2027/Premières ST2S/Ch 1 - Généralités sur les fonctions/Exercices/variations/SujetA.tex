\noindent\textbf{Exercice 1}\quad En utilisant la définition de la croissance ou de la
décroissance d'une fonction, déterminer pour chaque fonction si elle est croissante ou
décroissante sur l'intervalle indiqué.

\medskip
\noindent\begin{tabular}{@{}p{0.47\linewidth}@{}p{0.47\linewidth}@{}}
1. $f(x)=3x+2$ sur $\mathrm{I}=\mathbb{R}$.
  & 2. $f(x)=-5x^{2}-3$ sur $\mathrm{I}=[0\,;+\infty[$ \\[0.9em]
3. $f(x)=4x^{2}+1$ sur $\mathrm{I}=\,]-\infty\,;0]$
  & 4. $f(x)=4(x+2)^{2}-3$ sur $\mathrm{I}=\,]-\infty\,;-2]$
\end{tabular}

\noindent\textbf{Exercice 4}\quad On donne ci-dessous les tableaux de variations de deux
fonctions $f$ et $g$.

\medskip
\noindent
\begin{minipage}{0.48\linewidth}
\centering
\begin{tikzpicture}[scale=0.85]
  \tkzTabInit[lgt=1.4,espcl=1.6]{$x$/1,$f(x)$/2}{$-4$,$-3$,$-1$,$2$,$3$}
  \tkzTabVar{-/$-2$,+/$0$,-/$-3$,+/$4$,-/$2$}
\end{tikzpicture}
\end{minipage}
\hfill
\begin{minipage}{0.48\linewidth}
\centering
\begin{tikzpicture}[scale=0.85]
  \tkzTabInit[lgt=1.4,espcl=1.6]{$x$/1,$g(x)$/2}{$-5$,$-2$,$0$,$3$,$5$}
  \tkzTabVar{+/$0$,-/$-3$,+/$2$,-/$1$,+/$3$}
\end{tikzpicture}
\end{minipage}

\medskip
\begin{enumerate}[leftmargin=*,itemsep=0.5em]
  \item Comparer, si c'est possible, les images suivantes avec un signe d'inégalité.
  \medskip

  \noindent\hspace*{-1.5em}\begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}llll@{}}
    a. $f(-4)$ \rule{0.9cm}{0.4pt} $f(-3)$
      & b. $f(0)$ \rule{0.9cm}{0.4pt} $f(1)$
      & c. $f(2,\!5)$ \rule{0.9cm}{0.4pt} $f(-2)$
      & d. $f(-3,\!5)$ \rule{0.9cm}{0.4pt} $f(2,\!1)$ \\[1em]
    e. $g(1)$ \rule{0.9cm}{0.4pt} $g(-4)$
      & f. $g(-1)$ \rule{0.9cm}{0.4pt} $g(1)$
      & g. $g(-4)$ \rule{0.9cm}{0.4pt} $g(-3)$
      & h. $g(1)$ \rule{0.9cm}{0.4pt} $g(4)$
  \end{tabular*}

  \item Tracer une représentation possible des fonctions $f$ et $g$.
\end{enumerate}
