\section{Définition et notations}

\begin{definition}{}{}
  \begin{multihide}
  Une \textbf{fonction} $f$ est la donnée de deux choses 
  \begin{itemize}
    \item Un \textbf{ensemble de définition} $D_f$. $D_f$ est le plus souvent un intervalle ou une union d'intervalles de \setR.
    \item pour chaque nombre $x$ de $D_f$, un nombre $f(x)$ appelé $\textbf{image}$ de x. $x$ est alors appelée l'\textbf{antécédent} de $f(x)$.
  \end{itemize}
 \end{multihide}
\end{definition}

\vfill

\begin{example}
  Dire que : $f(\textcolor{orange}{2})=\textcolor{green!50!black}{5}$ signifie que :

\begin{center}
\begin{tikzpicture}[>=stealth]
    % Association
    \node (deux) at (0,0) {\textcolor{orange}{$2$}};
    \node (cinq) at (2,0) {\textcolor{green!50!black}{$5$}};
    \draw[|->] (deux.east) -- (cinq.west) node[above,midway] {$f$};

    % Étiquettes
    \node[
        fill=orange!15,
        minimum width=4.2cm,
        minimum height=0.65cm
    ] (E) at (-1.5,-1.1) {\textcolor{orange!80!black}{Antécédent de 5}};

    \node[
        fill=green!15,
        minimum width=2.8cm,
        minimum height=0.65cm
    ] (F) at (3,-1.1) {\textcolor{green!50!black}{Image de 2}};

    % Flèches vers les étiquettes
    \draw[->, orange] (E) -- (deux);
    \draw[->, green!50!black] (F) -- (cinq);
\end{tikzpicture}

\end{center}

On dit que l'\textbf{\hide{image} de \textcolor{orange}{2}} par la fonction $f$ est \textcolor{green!50!black}{5} et un \textbf{\hide{antécédent} de \textcolor{green!50!black}{5}} par $f$ est \textcolor{orange}{2}.
\end{example}

\vfill

\begin{remark}
  Un nombre peut avoir aucun, un seul ou plusieurs antécédents.
  En revanche, un nombre de l'ensemble de définition $D_f$ \underline{n'a qu'une seule image}.
  Un nombre hors de $D_f$ n'en a pas.%et un nombre qui n'est pas dans l'ensemble de définition n'a pas d'image.
\end{remark}

\vfill

\begin{example}
Soit $f$ une fonction d'ensemble de définition $D_f = [1,5\,;\,6]$.

On souhaite donner des images aux nombres de l'intervalle $[1,5\,;\,6]$.
%En revanche, on ne donne pas d'image aux nombres hors de $D_f$. Ainsi, écrire $f(1)$ ou $f(6,5)$ n'a aucun sens pour cette fonction.

On peut définir en partie $f$ à l'aide d'un tableau comme celui-ci :
    
\begin{center}
\begin{tabularx}{\linewidth}{|Y|Y|Y|Y|Y|Y|Y|Y|Y|}
\hline
$x$ & $1,5$ & $2$ & $2,5$ & $3$ & $3,5$ & $4$ & $5$ & $6$ \\
\hline
$f(x)$ & $5,25$ & $3$ & $1,25$ & $0$ & $-0,75$ & $-1$ & $0$ & $\dots$ \\
\hline
\end{tabularx}
\end{center}

On peut voir grâce à ce tableau que :

\begin{tasks}[label=\textbullet](3)
    \task $f(1,5) = \dots$
    \task $f(3) = \dots$
    \task $f(4) = \dots$
    \task $f(\dots) = -1$
    \task $f(\dots) = 1,25$
    \task $f(\dots) = 0$
\end{tasks}

On dit par exemple que $5,25$ est $\dots \dots \dots$ de $\dots$ par $f$ et que $1,5$ est $\dots \dots \dots$ de $\dots$ par $f$. $0$ a plusieurs antécédents qui sont $\dots$ et $\dots$.

Avec un tableau, on ne peut pas donner toutes des images à tous les nombres de l'intervalle $[1,5\,;\,6]$.
%Il faudrait donc une infinité de cases dans ce tableau pour qu'il définisse complètement la fonction $f$.
Lorsque c'est possible, on peut aussi donner une formule.

Ici, on donne la formule suivante :

$$f(x) = x^2 - 8x + 15$$

On peut alors calculer l'image d'un nombre de $[1,5\,;\,6]$ en remplaçant $x$ par ce nombre dans la formule.
\newpage 

Par exemple, si on veut calculer l'image de $2$ par $f$, on écrit :
\[
f(2) = 2^2 - 8 \times 2 + 15 = 4 - 16 + 15 = 3
\]
et on retrouve bien que l'image de $2$ par $f$ est $3$.
De la même manière :
$$f(6) = \dots$$
\end{example}

\begin{definition}{Représentation $C_f$ de $f$}{}
  \begin{multihide}
    La représentation graphique (ou courbe) $C_f$ d'une fonction $f$ dans un repère est l'ensemble des points du repère de coordonnées $(x ; f(x))$ pour les $x$ de l'ensemble de définition $D_f$.

\begin{proposition}{}{}
  \begin{multihide}
    Un point $A$ de coordonnées $(x_A; y_A)$ appartient à la courbe de $f$ si et seulement si $x_A \in D_f$ et $f(x_A) = y_A$.
  \end{multihide}
\end{proposition}

\section{Variations d'une fonction}

\begin{definition}{Sens de variation d'une fonction}{}
  \begin{multihide}
  Soit $f$ définie sur un intervalle $I$, $f$ est 
  \begin{itemize}
    \item \textbf{croissante}, si pour tout réels $a$ et $b$ de $I$ tels que $a < b$ alors $f(a) \le f(b)$,
    \item \textbf{décroissante}, si pour tout réels $a$ et $b$ de $I$ tels que $a < b$ alors $f(a) \ge f(b)$,
    \item \textbf{constante}, si on a toujours $f(a) = f(b)$.
  \end{itemize}
  \end{multihide}
\end{definition}

  \node[myblue, below] at (\ax,0) {$a$};
  \node[myblue, below] at (\bx,0) {$b$};

  \node[below] at (3, -1) {$f$ croissante sur {\color{myblue}$I$}};
\end{scope}

% ----------- Panneau droit : f décroissante sur [a,b] -----------
\begin{scope}[xshift=9cm]
  \draw[myblue, -]  (-0.5,0) -- (3,0);
  \draw[myorange, ->]  (3,0) -- (6.5,0);
  \draw[->] (0,-0.5) -- (0,4.5);

  \draw[myblue, very thick, domain=0.5:3, samples=60, smooth] plot (\x,{f(\x)});
  \draw[myorange, very thick, domain=3:5.5, samples=60, smooth] plot (\x,{f(\x)});

  \def\ax{4} \def\bx{5}
  \pgfmathsetmacro{\fa}{f(\ax)}
  \pgfmathsetmacro{\fb}{f(\bx)}
  \draw[myorange, dashed] (\ax,0) -- (\ax,\fa) -- (0,\fa);
  \draw[myorange, dashed] (\bx,0) -- (\bx,\fb) -- (0,\fb);
  \node[myorange, left]  at (0,\fa) {$f(a)$};
  \node[myorange, left]  at (0,\fb) {$f(b)$};
  \node[myorange, below] at (\ax,0) {$a$};
  \node[myorange, below] at (\bx,0) {$b$};
  
  \node[below] at (3, -1) {$f$ décroissante sur {\color{myorange}$J$}};
\end{scope}

\end{tikzpicture}
\end{center}

\begin{remarks}
  \begin{itemize}
    \item[]
    \item Une fonction est \textbf{strictement} croissante (respectivement décroissante) sur $I$ si, pour tous nombres $a$ et $b$ de $I$ tels que $a < b$, alors \hide{$f(a) < f(b)$} (respectivement \hide{$f(a) > f(b)$}).
    \item Une fonction est \textbf{monotone} sur $I$ si elle garde le même sens de variation sur $I$.
  \end{itemize}
\end{remarks}

  On peut présenter les variations d'une fonction $f$ par un \textbf{tableau de variations}.
