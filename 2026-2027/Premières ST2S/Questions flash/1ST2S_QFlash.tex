\begin{prop}{1. Calcul numérique}
\qr{Calculer $\sqrt{3} \times \sqrt{12}$.}{1.3em}{$\sqrt{3} \times \sqrt{12} = \sqrt{36} = 6$}
\end{prop}

\begin{prop}{2. Calcul littéral}
\qr{Résoudre l'équation\\ $(x - 4)(2x + 6) = 0$.}{2.4em}{$x - 4 = 0$ ou $2x + 6 = 0$\\ donc $x = 4$ ou $x = -3$.}
\end{prop}

\begin{prop}{3. Évolutions successives}
\qr{Un article coûte $100$ €.\\ Son prix augmente de $20\,\%$,\\ puis il baisse de $20\,\%$.\par\vspace{16pt}Calculer son prix final.\\ En déduire l'évolution du prix, en $\%$.}{3.6em}{$100 \times 1{,}2 = 120$ €\\ $120 \times 0{,}8 = 96$ €\\ Le prix a baissé de $4\,\%$.}
\end{prop}
