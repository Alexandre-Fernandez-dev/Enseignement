\textbf{Nom et prénom :}\enspace\dotfill

\begin{exercice}{}{}
Calculer et donner le résultat sous la forme d'une fraction irréductible.
\begin{tasks}(3)
  \task $\dfrac{1}{3} + \dfrac{8}{3} \times \dfrac{3}{4}$
  \task $\dfrac{4}{7}\div\dfrac{1}{6}$
  \task $\dfrac{4}{5}-\dfrac{1}{6}$
\end{tasks}
\end{exercice}

\begin{exercice}{}{}
  Calculer les puissances suivantes :
 \begin{tasks}(2)
  \task $3^3$
  \task $\dfrac{4^{4}}{4^{2}}$%Soit $a$ un nombre réel non nul et $n$ un nombre entier relatif. 
 \end{tasks}
\end{exercice}


\begin{exercice}{}{}
  \begin{enumerate}
    \item Compléter avec le signe $<$ ou $>$.
  $$ -7,2 \quad \ldots\ldots   \quad-7,6 $$
  
    \item Calculer.


    \begin{tasks}(3)
      \task $ -6  \times 3$
      \task $ 2 - (-2)$
      \task $ -8 \div (-4)$
    \end{tasks}

  \end{enumerate}
\end{exercice}

\begin{exercice}{}{}
  \begin{enumerate}
    \item Résoudre l'équation $3x - 3 = 13 - x$
    \item Développer et réduire.
    \begin{tasks}(2)
      \task $(-9z - 3) \times 10$
      \task $(8z + 5) (z - 7)$
    \end{tasks}
    \item Factoriser les expressions suivantes :
    \begin{tasks}(2)
      \task $12n  + 14c$
      \task $x(2x + 4) - 2(2x + 4)$
    \end{tasks}
  \end{enumerate}
\end{exercice}
