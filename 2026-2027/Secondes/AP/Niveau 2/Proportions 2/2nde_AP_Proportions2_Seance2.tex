\begin{tcolorbox}[colback=white, colframe=couleur_theme, title=\textbf{Rappel : calculer le total connaissant une partie}, fonttitle=\bfseries]
On connaît la partie et la proportion, on cherche le total. On revient à $10\,\%$ (ou à $1\,\%$), puis on remonte à $100\,\%$.

\smallskip
\textbf{Exemple} : $20\,\%$ des élèves d'un club, soit $6$ élèves, sont en Seconde. \quad
$20\,\% \to 6$, \quad donc $10\,\% \to 3$, \quad donc $100\,\% \to 30$. \quad Le club compte $30$ élèves.

\smallskip
\textbf{Avec une fraction} : $25\,\% = \dfrac{1}{4}$. Si un quart du total vaut $12$, le total vaut $4 \times 12 = 48$.

\smallskip
\textbf{Vérifier} : on calcule $p \times$ total et on doit retrouver la partie : \quad $20\,\%$ de $30$ \quad $= 6$.
\end{tcolorbox}

\textbf{Pour aller plus loin.} Un lycée compte $800$ élèves. $75\,\%$ des élèves mangent à la cantine. Parmi eux, $20\,\%$ choisissent le menu végétarien.
\begin{enumerate}[itemsep=0.3em]
	\item Combien d'élèves mangent à la cantine~?
	\ligne\ligne
	\item Combien d'élèves choisissent le menu végétarien~?
	\ligne\ligne
	\item Quelle est la proportion d'élèves du lycée qui choisissent le menu végétarien~?
	\ligne\ligne
	\item Calculer $0{,}75 \times 0{,}2$. Que remarque-t-on~?
	\ligne\ligne
\end{enumerate}
\end{EXO}

\begin{EXO}{}{}
\begin{enumerate}[label=\textbf{\alph*.}]
	\item Proportion : $\dfrac{10}{40} = \res{25\,\%}$.
	\item Partie : $10\,\%$ de $50$~cL $= \res{5~\text{cL}}$.
	\item Total : $30\,\% \to 240$, $10\,\% \to 80$, $100\,\% \to \res{800}$ élèves.
	\item Proportion : $\dfrac{15}{250} = \dfrac{6}{100} = \res{6\,\%}$.
	\item Partie : $10\,\% \to 60$, $40\,\% \to 240$, $5\,\% \to 30$, donc $45\,\% \to \res{270}$ personnes.
	\item Total : $75\,\% = \dfrac{3}{4}$ ; $25\,\% \to 5$~h, donc $100\,\% \to \res{20~\text{h}}$.
\end{enumerate}
\end{EXO}

\begin{EXO}{}{}
Seconde A : $\dfrac{12}{30} = \dfrac{40}{100} = 40\,\%$. \quad Seconde B : $\dfrac{9}{20} = \dfrac{45}{100} = 45\,\%$.
Tom a tort : il y a plus de sportifs en Seconde A, mais la \textbf{proportion} est plus grande en Seconde B.
\end{EXO}

\begin{EXO}{}{}
\begin{enumerate}
	\item $75\,\%$ de $800$ $= \dfrac{3}{4} \times 800 = \res{600}$ élèves.
	\item $20\,\%$ de $600$ $= \res{120}$ élèves.
	\item $\dfrac{120}{800} = \dfrac{15}{100} = \res{15\,\%}$.
	\item $0{,}75 \times 0{,}2 = 0{,}15 = 15\,\%$ : on retrouve la proportion de la question 3. On multiplie les deux proportions.
\end{enumerate}
\end{EXO}
