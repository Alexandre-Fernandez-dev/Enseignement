\seance{Séance 1 : proportion et pourcentages simples}

\begin{tcolorbox}[colback=white, colframe=couleur_theme, title=\textbf{Rappel}, fonttitle=\bfseries]
\textbf{Proportion} \quad $= \dfrac{\text{partie}}{\text{total}}$. \qquad Exemple : $5$ élèves sur $20$ ont des lunettes. \quad Proportion : $\dfrac{5}{20} = \dfrac{1}{4}$.
\textbf{Pourcentage} : $25\,\%$ veut dire $25$ sur $100$, c'est-à-dire $\dfrac{25}{100}$.

\smallskip
\textbf{À connaître} : $50\,\% = \dfrac{1}{2}$ : on divise par~$2$ ; \enspace $25\,\% = \dfrac{1}{4}$ : on divise par~$4$ ; \enspace $10\,\% = \dfrac{1}{10}$ : on divise par~$10$.

\smallskip
$75\,\% = \dfrac{3}{4}$ : on divise par $4$, puis on multiplie par $3$. \qquad Exemple : $75\,\%$ de $80$ \quad $= 80 \div 4 \times 3 = 20 \times 3 = 60$.
\end{tcolorbox}
