\begin{EXO}{}{}
Calculer en respectant les priorités opératoires.

\begin{tasks}(2)
	\task $A=(-2)^3-3\times(-4)+(-1)^{2027}$
	\task $B=\dfrac{-3^2+(-3)^2}{(-5)^2-5^2+1}$
	\task $C=-2\times\left[3-(-4)^2\div(-8)\right]^2$
\end{tasks}
\end{EXO}

\begin{EXO}{}{}
\begin{enumerate}
	\item Sans calculer, donner le signe de chaque nombre. Justifier.
	\begin{tasks}(3)
		\task $(-3)^{17}\times(-5)^{12}$
		\task $\dfrac{(-2)^{9}\times(-7)^{3}}{-4^{6}}$
		\task $-(-2)^{8}\times(-1)^{2026}$
	\end{tasks}
	\item Calculer $S=(-1)^1+(-1)^2+(-1)^3+\dots+(-1)^{100}$, puis $T=(-1)^1+(-1)^2+\dots+(-1)^{101}$.
\end{enumerate}
\end{EXO}

\begin{EXO}{}{}
\begin{enumerate}
	\item Écrire sous la forme $a^n$, puis calculer :
	\quad a) $\dfrac{8^3\times 4^{-2}}{2^5}$ \qquad b) $\dfrac{27^2\times 9^{-3}}{3^{-4}}$
	\item Écrire sous la forme $2^p\times 3^q$ ($p$, $q$ entiers relatifs) :
	\quad a) $\dfrac{6^5}{2^3\times 3^4}$ \qquad b) $\dfrac{12^{3}}{18^{2}}$
\end{enumerate}
\end{EXO}

\begin{EXO}{}{}
Simplifier les expressions suivantes, où $x$ est un réel non nul.

\begin{tasks}(4)
	\task $x^3\times x^{-5}\times x^4$
	\task $\dfrac{\left(x^2\right)^3}{x^{-1}\times x^4}$
	\task $\left(-2x^3\right)^2\times(-x)^3$
	\task $\dfrac{\left(3x^2\right)^3}{9x^4}$
\end{tasks}
\end{EXO}

\begin{EXO}{}{}
Calculer sans calculatrice.

\begin{tasks}(2)
	\task $\dfrac{2^{2027}+2^{2026}}{2^{2025}}$
	\task $\dfrac{3^{20}-3^{18}}{8\times 3^{17}}$
\end{tasks}

\textit{Indication : pour b), on peut remarquer que $3^{20}=3^{18}\times 3^2$.}
\end{EXO}

\begin{EXO}{}{}
\begin{enumerate}
	\item Écrire $2^{300}$ sous la forme d'une puissance de $8$, et $3^{200}$ sous la forme d'une puissance de $9$.
	\item En déduire lequel des deux nombres $2^{300}$ et $3^{200}$ est le plus grand.
	\item De la même façon, comparer $2^{50}$ et $5^{20}$.
\end{enumerate}
\end{EXO}

\begin{EXO}{}{}
Une feuille de papier a une épaisseur de $0{,}1$~mm. On la plie en deux, puis encore en deux, et ainsi de suite (on suppose que c'est toujours possible).
\begin{enumerate}
	\item Quelle est l'épaisseur obtenue après $1$, $2$ puis $3$ pliages~? Justifier qu'après $n$ pliages, l'épaisseur est $0{,}1\times 2^n$~mm.
	\item Calculer l'épaisseur après $10$ pliages, sachant que $2^{10}=1\,024$.
	\item On utilise l'approximation $2^{10}\approx 10^3$. Montrer qu'après $40$ pliages, l'épaisseur est d'environ $10^{11}$~mm, puis convertir en kilomètres.
	\item La distance de la Terre à la Lune est d'environ $384\,000$~km. Combien de pliages suffiraient pour l'atteindre~?
\end{enumerate}
\end{EXO}
