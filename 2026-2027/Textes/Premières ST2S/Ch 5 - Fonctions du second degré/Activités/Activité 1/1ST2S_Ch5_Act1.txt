\begin{objectif}
Découvrir une fonction du second degré donnée sous forme factorisée : ses racines et sa courbe, la parabole.
\end{objectif}

\section*{Le saut du dauphin}

Dans un bassin, à l'instant $t = 0$, un soigneur lance un poisson. Le dauphin sort de l'eau $1$ seconde plus tard. Il attrape le poisson à l'instant $t = 2$ secondes, à une hauteur de $2{,}5$ mètres, puis replonge dans l'eau à l'instant $t = 7$ secondes.

\begin{center}
\begin{tikzpicture}[x=1.3cm, y=0.62cm]
\fill[blue!12] (-0.2,0) rectangle (8,-1.3);
\draw[blue!60, thick] (-0.2,0) -- (8,0);
\node[blue!70!black, font=\small] at (4.8,-0.65) {surface de l'eau : hauteur $0$};
\draw[thick, dashed, blue!70!black, domain=1:7, samples=80] plot (\x,{-0.5*(\x-1)*(\x-7)});
\draw[dotted, thick] (2,0) -- (2,2.5);
\node[font=\small, right] at (2,1.25) {$2{,}5$ m};
\poisson{2.05}{2.68}
\draw[->, gray] (0.9,3.6) node[left, black, font=\small] {poisson} .. controls (1.4,3.6) .. (1.85,2.85);
\dauphin{0.55}{-0.95}{55}{0.55}
\foreach \t/\lab in {1/{$t = 1$}, 2/{$t = 2$}, 7/{$t = 7$}} {
\fill (\t,0) circle (1.6pt);
\node[font=\small, below] at (\t,-1.3) {\lab};
\draw[gray, very thin] (\t,0) -- (\t,-1.3);
}
\end{tikzpicture}
\end{center}

On note $f(t)$ la hauteur du dauphin, en mètres, au-dessus de l'eau à l'instant $t$, en secondes, pour $t$ entre $1$ et $7$. La surface de l'eau correspond à la hauteur $0$.

On modélise la trajectoire par une fonction de la forme
\[ f(t) = a\,(t - t_1)(t - t_2), \]
où $a$, $t_1$ et $t_2$ sont des nombres à déterminer, avec $a \neq 0$.

\begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=1.3em]
\item À quels instants le dauphin est-il au niveau de l'eau~? Que vaut alors $f(t)$~?
\reponse{1}

\item Que vaut $f(t_1)$~? Et $f(t_2)$~? En déduire les valeurs de $t_1$ et $t_2$, puis compléter :
\[ f(t) = a\,(t - \trou{1cm})(t - \trou{1cm}). \]
\reponse{1}

\item Le dauphin est à $2{,}5$ m de hauteur à l'instant $t = 2$ : on a donc $f(2) = 2{,}5$. En déduire la valeur de $a$.
\reponse{2}

\item On admet que $f(t) = -0{,}5\,(t - 1)(t - 7)$.
\begin{enumerate}[label=\textbf{\alph*)}, leftmargin=1.5em]
\item Compléter le tableau de valeurs.
\begin{center}
\renewcommand{\arraystretch}{1.8}
\begin{tabular}{|c|*{7}{>{\centering\arraybackslash}p{1.2cm}|}}
\hline
$t$ & $1$ & $2$ & $3$ & $4$ & $5$ & $6$ & $7$ \\
\hline
$f(t)$ & & & & & & & \\
\hline
\end{tabular}
\end{center}
\item Placer les points dans le repère. Le professeur trace la courbe avec GeoGebra : la reproduire. Cette courbe s'appelle une \textbf{parabole}.
\end{enumerate}
\begin{center}
\begin{tikzpicture}
\begin{axis}[
width=0.75\textwidth, height=6cm,
xmin=0, xmax=7.5, ymin=0, ymax=5,
xtick={0,...,7}, ytick={0,...,5}, minor x tick num=1, minor y tick num=1,
grid=both, major grid style={gray!45}, minor grid style={gray!20},
axis lines=left, xlabel={$t$ (s)}, ylabel={hauteur (m)},
ylabel style={font=\small}, xlabel style={at={(1,0)}, anchor=west},
]
\end{axis}
\end{tikzpicture}
\end{center}

\item Développer l'expression $f(t) = -0{,}5\,(t - 1)(t - 7)$.
\reponse{2}
\end{enumerate}

\begin{bilan}
\begin{itemize}[leftmargin=1.2em]
\item Une fonction de la forme $f(x) = a\,(x - x_1)(x - x_2)$, avec $a \neq 0$, est une fonction du second degré écrite sous \textbf{forme factorisée}. Sa courbe est une \trou{2.5cm}.
\item Les nombres $x_1$ et $x_2$ sont les \textbf{racines} de $f$ : ce sont les solutions de l'équation \mbox{$f(x) = \trou{1cm}$}.
\item En développant, on obtient la \textbf{forme développée} $f(x) = ax^2 + bx + c$. Pour le dauphin : $a = \trou{1.2cm}$, $b = \trou{1.2cm}$ et $c = \trou{1.2cm}$. Le coefficient $a$ est le même dans les deux formes.
\end{itemize}
\end{bilan}
